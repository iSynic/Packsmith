cmake_minimum_required(VERSION 3.24)
project(Packsmith VERSION 0.1.0 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "This prototype is currently qualified for Windows only")
endif()
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.10.2 EXACT REQUIRED COMPONENTS Core Gui Widgets)
set(SEVENZIP_SOURCE "${CMAKE_SOURCE_DIR}/assessment/references/sevenzip" CACHE PATH "Pinned 7-Zip SDK checkout")
if(NOT EXISTS "${SEVENZIP_SOURCE}/CPP/7zip/Archive/IArchive.h")
  message(FATAL_ERROR "Acquire the pinned assessment reference checkout first")
endif()
add_executable(packsmith-worker app/worker.cpp)
target_include_directories(packsmith-worker PRIVATE "${SEVENZIP_SOURCE}/CPP")
target_link_libraries(packsmith-worker PRIVATE Qt6::Core oleaut32 ole32 uuid)
target_compile_definitions(packsmith-worker PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_executable(Packsmith WIN32 app/main.cpp app/packsmith.rc)
qt_add_resources(Packsmith brand PREFIX "/brand" BASE assets/icons FILES
  assets/icons/packsmith.ico assets/icons/packsmith-archive.ico)
target_include_directories(Packsmith PRIVATE "${CMAKE_SOURCE_DIR}")
target_link_libraries(Packsmith PRIVATE Qt6::Widgets)
target_compile_definitions(Packsmith PRIVATE UNICODE _UNICODE NOMINMAX)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  target_compile_options(Packsmith PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(packsmith-worker PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
  add_test(NAME worker-behavior COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/worker_test.py" "${CMAKE_SOURCE_DIR}/dist/Packsmith-preview/workers/packsmith-worker.exe")
endif()
